function packageEvidence(specifier, expectedName) {
	const entry = fs.realpathSync(requireTool.resolve(specifier));
	let directory = path.dirname(entry);
	for (;;) {
		const manifest = path.join(directory, 'package.json');
		if (fs.existsSync(manifest)) {
			const source = readInput(manifest);
			const data = JSON.parse(source);
			if (data.name === expectedName)
				return {
					name: data.name,
					version: data.version,
					entry,
					entrySha256: hash(readInput(entry)),
					root: directory,
					manifestSha256: hash(source),
				};
		}
		const parent = path.dirname(directory);
		assert.notEqual(parent, directory, 'Missing dependency manifest: ' + expectedName);
		directory = parent;
	}
}

function compilerSnapshot(root) {
	const directory = path.join(root, 'packages/octane/src/compiler');
	const files = fs
		.readdirSync(directory, { recursive: true })
		.filter((name) => /\.[cm]?js$/.test(name));
	return files.sort().map((name) => {
		const file = path.join(directory, name);
		const source = readInput(file);
		return { path: path.relative(root, file), sha256: hash(source) };
	});
}

function dataFor(entry) {
	if (!entry.scenario.reads)
		return {
			props: { tick: 0, value: 7 },
			value: () => '7',
			update: null,
			dispose() {},
		};
	const scope = entry.api.createScope({ scopeKey: 'native-cost-' + entry.name });
	const sources$ = Array.from({ length: 16 }, (_, index) => scope.signal$('source-' + index, 7));
	let current = 7;
	return {
		props: { tick: 0, source$: sources$[0], sources$ },
		value: () => String(current * entry.scenario.reads),
		update(value) {
			current = value;
			scope.batch(() => {
				for (const source$ of entry.scenario.distinct ? sources$ : [sources$[0]])
					source$.set(value);
			});
		},
		dispose() {
			scope.dispose();
		},
	};
}

	return output;
}

          const expectedName=expectedLabel+record.name.slice(record.name.indexOf(":"));
          const old=reference.compiled.find((item)=>item.name===expectedName);
          assert.ok(old);
          assert.equal(record.bundleSha256,old.bundleSha256,'Frozen source must produce the exact existing measured bundle');
          record.matchesExistingBundle=true;
        }
        pair.push(prepare(entry));
      }
      assert.deepEqual(pair.map((entry)=>entry.entry.target),['baseline','current']);
      const name=scenario.name+(enabled?':native':':ordinary');
      cases.push({name,pair});samplesByCase.set(name,[]);
    }
  }
  for(let warmup=0;warmup<warmups;warmup++){
    for(const scenario of shuffled(cases)){
      const order=warmup%2===0?[0,1]:[1,0];
      const records=order.map((side)=>measure(scenario.pair[side]));
      payload.warmupRecords.push({warmup,case:scenario.name,records});
    }
    await nextTurn();
    console.log('Focused TSRX warmup '+(warmup+1)+'/'+warmups+' complete; output and identity controls passed.');
  }
  for(let round=0;round<samples;round++){
    for(const scenario of shuffled(cases)){
      const order=nextRandom()%2===0?[0,1,1,0]:[1,0,0,1];
      const records=order.map((side)=>measure(scenario.pair[side]));
      const byTarget=(target,metric)=>records.filter((row)=>row.target===target).reduce((sum,row)=>sum+row[metric],0)/2;
      const row={index:pairIndex++,round,case:scenario.name,order:order.map((side)=>side===0?'baseline':'current'),records,ratio:{}};
      for(const metric of ['wall_us','cpu_us','main_thread_cpu_us']){
        if(records.some((record)=>record[metric]===null))continue;
        row.ratio[metric]=byTarget('current',metric)/byTarget('baseline',metric);
      }
      payload.pairs.push(row);samplesByCase.get(scenario.name).push(row);
      for(const prepared of scenario.pair){verifySignal(prepared,round+8);verifySignal(prepared,7);}
      await nextTurn();
    }
    if((round+1)%5===0||round===samples-1)console.log('Focused TSRX pair round '+(round+1)+'/'+samples+' complete; native signal controls passed.');
  }
  for(const scenario of cases){
    const rows=samplesByCase.get(scenario.name);
    const summary={case:scenario.name,pairs:rows.length,ratios:{},absolute:{}};
    for(const metric of ['wall_us','cpu_us','main_thread_cpu_us']){
      if(rows.some(row=>row.ratio[metric]===undefined))continue;
      summary.ratios[metric]=ratioSummary(rows.map((row)=>row.ratio[metric]));
      summary.absolute[metric]={};
      for(const target of ['baseline','current'])summary.absolute[metric][target]=timingStatForJson(summarizeSamples(rows.flatMap(row=>row.records.filter(record=>record.target===target).map(record=>record[metric])),{scoreMode:'mean'}));
    }
    payload.summaries.push(summary);
  }
  for(const scenario of cases)for(const prepared of scenario.pair){
    prepared.root.unmount();assert.equal(prepared.container.textContent,'');
    prepared.data.update?.(999);
    assert.equal(prepared.container.textContent,'','A live producer must not revive a retired root');
    prepared.container.remove();prepared.data.dispose();
  }
  await nextTurn();
  for(const row of payload.pairs)for(const record of row.records){
    const events=gcEvents.filter((event)=>event.startTime<record.end&&event.startTime+event.duration>record.start);
    record.gcEvents=events.length;record.gcWallMs=events.reduce((sum,event)=>sum+Math.min(record.end,event.startTime+event.duration)-Math.max(record.start,event.startTime),0);
  }
  assert.equal(hashFile(fixtureFile),hash(fixture),'Authored fixture changed during focused measurement');
  for(const[file,contents]of snapshotInputs)assert.equal(hashFile(file),hash(contents),'Source changed during focused measurement: '+file);
  for(const target of roots){const ref=sourceRefs[target.label];if(ref===null)continue;for(const[file,contents]of snapshotInputs){if(!file.startsWith(path.join(target.root,'packages/octane')+path.sep))continue;const relative=path.relative(target.root,file).replaceAll('\\','/');assert.equal(hash(contents),hash(git('show',ref+':'+relative)),'Archived package bytes differ from Git');}}
  payload.sourceUnchanged=true;payload.status='passed';
}catch(error){failure=error instanceof Error?error.stack:String(error);payload.failed=failure;console.error(failure);}
finally{
  observer.disconnect();
  payload.gcEvents=gcEvents;
  payload.finishedAt=new Date().toISOString();
  payload.hostAfter={loadAverage:os.loadavg(),freeMemory:os.freemem(),resourceUsage:process.resourceUsage()};
  payload.sourceInputs=[...snapshotInputs].map(([file,contents])=>({file,sha256:hash(contents)}));
  await window.happyDOM.close();
  assert.ok(process.env.BENCH_JSON,'Focused evidence requires a fresh BENCH_JSON path');
  assert.equal(fs.existsSync(process.env.BENCH_JSON),false,'Never overwrite focused evidence');
  fs.writeFileSync(process.env.BENCH_JSON,JSON.stringify(payload,null,2)+'\n');
}
if(failure)process.exitCode=1;
